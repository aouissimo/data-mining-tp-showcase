Introduction

This project presents a robust implementation of the ID3 (Iterative Dichotomiser 3) algorithm, integrated into a premium interactive dashboard. As the core of the Knowledge Discovery in Databases (KDD) process, the application transforms raw, unstructured data into actionable intelligence in the form of transparent decision trees.

Key focus areas included algorithmic integrity, data preprocessing, and visual clarity using modern web technologies.

Recursive Methodology

The ID3 algorithm utilizes Information Gain calculated via Shannon Entropy to recursively partition the dataset. At each step, the attribute that reduces the most "uncertainty" in the target label is selected as the splitting node.

Entropy(S) = - ∑ pi log2(pi)

Case Study: Weather Dataset

  • 1. The algorithm calculates the base entropy of the 14 records (Play = Yes/No).
  • 2. It iterates through Outlook, Temperature, Humidity, and Wind.
  • 3. Outlook yields the highest Information Gain (~0.247) and becomes the root.
  • 4. For Outlook = Overcast, entropy is 0 (Pure Node), so it immediately returns RESULT: YES.

TypeScript ID3 Core src/utils/id3-core.ts 
/**
 * ID3 (Iterative Dichotomiser 3) Core Algorithm
 * Specifically designed for categorical data.
 */

import type { DataRow } from './preprocessing';

export interface TreeNode {
  attribute?: string;
  value?: string | number;
  children: { [key: string]: TreeNode };
  isLeaf: boolean;
  label?: string;
}

export const calculateEntropy = (data: DataRow[], targetAttr: string): number => {
  const total = data.length;
  if (total === 0) return 0;

  const counts: { [key: string]: number } = {};
  data.forEach((row) => {
    const label = row[targetAttr];
    counts[label] = (counts[label] || 0) + 1;
  });

  return Object.values(counts).reduce((acc, count) => {
    const p = count / total;
    return acc - p * Math.log2(p);
  }, 0);
};

export const calculateInfoGain = (data: DataRow[], attr: string, targetAttr: string): number => {
  const totalEntropy = calculateEntropy(data, targetAttr);
  const total = data.length;
  const attrValues = Array.from(new Set(data.map(row => row[attr])));
  let weightedEntropy = 0;

  attrValues.forEach((val) => {
    const subset = data.filter(row => row[attr] === val);
    const subsetEntropy = calculateEntropy(subset, targetAttr);
    weightedEntropy += (subset.length / total) * subsetEntropy;
  });

  return totalEntropy - weightedEntropy;
};

export const buildID3Tree = (
  data: DataRow[],
  targetAttr: string,
  remainingAttrs: string[]
): TreeNode => {
  const labels = Array.from(new Set(data.map(row => row[targetAttr])));
  if (labels.length === 1) {
    return { isLeaf: true, label: String(labels[0]), children: {} };
  }

  if (remainingAttrs.length === 0) {
    const counts: { [key: string]: number } = {};
    data.forEach(row => {
      const label = row[targetAttr];
      counts[label] = (counts[label] || 0) + 1;
    });
    const majorityLabel = Object.keys(counts).reduce((a, b) => (counts[a] > counts[b] ? a : b));
    return { isLeaf: true, label: String(majorityLabel), children: {} };
  }

  let bestAttr = remainingAttrs[0];
  let maxGain = -1;

  remainingAttrs.forEach((attr) => {
    const gain = calculateInfoGain(data, attr, targetAttr);
    if (gain > maxGain) {
      maxGain = gain;
      bestAttr = attr;
    }
  });

  const root: TreeNode = {
    attribute: bestAttr,
    children: {},
    isLeaf: false,
  };

  const attrValues = Array.from(new Set(data.map(row => row[bestAttr])));
  const nextRemainingAttrs = remainingAttrs.filter(a => a !== bestAttr);

  attrValues.forEach((val) => {
    const subset = data.filter(row => row[bestAttr] === val);
    if (subset.length === 0) {
      const counts: { [key: string]: number } = {};
      data.forEach(row => {
        const label = row[targetAttr];
        counts[label] = (counts[label] || 0) + 1;
      });
      const majorityLabel = Object.keys(counts).reduce((a, b) => (counts[a] > counts[b] ? a : b));
      root.children[String(val)] = { isLeaf: true, label: String(majorityLabel), children: {} };
    } else {
      root.children[String(val)] = buildID3Tree(subset, targetAttr, nextRemainingAttrs);
    }
  });

  return root;
};

export const predict = (tree: TreeNode, row: DataRow): string => {
  if (tree.isLeaf) return tree.label!;

  const attrVal = String(row[tree.attribute!]);
  const child = tree.children[attrVal];

  if (!child) {
    return "N/A";
  }

  return predict(child, row);
};

def calculate_entropy(data: list[DataRow], target_attr: str) -> float:
    total = len(data)
    if total == 0:
        return 0.0
    counts = Counter(row[target_attr] for row in data)
    entropy = 0.0
    for count in counts.values():
        p = count / total
        entropy -= p * math.log2(p)
    return entropy

def calculate_info_gain(data: list[DataRow], attr: str, target_attr: str) -> float:
    total_entropy = calculate_entropy(data, target_attr)
    total = len(data)
    attr_values = list(dict.fromkeys(row[attr] for row in data))
    weighted_entropy = 0.0
    for value in attr_values:
        subset = [row for row in data if row[attr] == value]
        weighted_entropy += (len(subset) / total) * calculate_entropy(subset, target_attr)
    return total_entropy - weighted_entropy

def majority_label(data: list[DataRow], target_attr: str) -> str:
    counts: dict[Any, int] = {}
    for row in data:
        label = row[target_attr]
        counts[label] = counts.get(label, 0) + 1
    winner = next(iter(counts))
    for label in counts:
        if counts[label] > counts[winner]:
            winner = label
    return str(winner)

def build_id3_tree(data: list[DataRow], target_attr: str, remaining_attrs: list[str]) -> dict[str, Any]:
    labels = list(dict.fromkeys(row[target_attr] for row in data))
    if len(labels) == 1:
        return {"isLeaf": True, "label": str(labels[0]), "children": {}}
    if not remaining_attrs:
        return {"isLeaf": True, "label": majority_label(data, target_attr), "children": {}}

    best_attr = remaining_attrs[0]
    max_gain = -1.0
    for attr in remaining_attrs:
        gain = calculate_info_gain(data, attr, target_attr)
        if gain > max_gain:
            max_gain = gain
            best_attr = attr

    root: dict[str, Any] = {"attribute": best_attr, "children": {}, "isLeaf": False}
    next_remaining_attrs = [attr for attr in remaining_attrs if attr != best_attr]

    for value in list(dict.fromkeys(row[best_attr] for row in data)):
        subset = [row for row in data if row[best_attr] == value]
        if not subset:
            root["children"][str(value)] = {
                "isLeaf": True,
                "label": majority_label(data, target_attr),
                "children": {},
            }
        else:
            root["children"][str(value)] = build_id3_tree(subset, target_attr, next_remaining_attrs)
    return root

def predict(tree: dict[str, Any], row: DataRow) -> str:
    if tree["isLeaf"]:
        return tree["label"]
    attr_value = str(row[tree["attribute"]])
    child = tree["children"].get(attr_value)
    if child is None:
        return "N/A"
    return predict(child, row)

def handle_missing_values_mode(data: list[DataRow]) -> list[DataRow]:
    result = deepcopy(data)
    columns = list(result[0].keys())
    for col in columns:
        values = [row[col] for row in data if row[col] not in (None, "")]
        counts: dict[Any, int] = {}
        for value in values:
            counts[value] = counts.get(value, 0) + 1
        replacement = next(iter(counts))
        for value in counts:
            if counts[value] > counts[replacement]:
                replacement = value
        for row in result:
            if row[col] in (None, ""):
                row[col] = replacement
    return result

def discretize_equal_width(data: list[DataRow], col: str, num_bins: int) -> list[DataRow]:
    values = [row[col] for row in data]
    min_value = min(values)
    max_value = max(values)
    width = (max_value - min_value) / num_bins
    result: list[DataRow] = []
    for row in data:
        value = row[col]
        bin_index = math.floor((value - min_value) / width)
        if bin_index >= num_bins:
            bin_index = num_bins - 1
        left = min_value + bin_index * width
        right = min_value + (bin_index + 1) * width
        bracket = "]" if bin_index == num_bins - 1 else ")"
        bin_range = f"[{left:.2f}, {right:.2f}{bracket}"
        result.append({**row, col: bin_range})
    return result
